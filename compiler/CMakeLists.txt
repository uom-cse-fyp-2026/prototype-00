# Build against exactly the MLIR used by torch-mlir, following mlir-quant-linear.
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}" "${LLVM_CMAKE_DIR}")
include(TableGen)
include(AddLLVM)
include(AddMLIR)
include(HandleLLVMOptions)
include_directories(${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS} "${GEN}")
link_directories(${LLVM_BUILD_LIBRARY_DIR})
add_definitions(${LLVM_DEFINITIONS})
get_property(dialect_libs GLOBAL PROPERTY MLIR_DIALECT_LIBS)
get_property(conversion_libs GLOBAL PROPERTY MLIR_CONVERSION_LIBS)
get_property(extension_libs GLOBAL PROPERTY MLIR_EXTENSION_LIBS)
set(NM_COMPILER_LIBS ${dialect_libs} ${conversion_libs} ${extension_libs} MLIROptLib MLIRPass MLIRTransforms)
foreach(lib MLIRRegisterAllDialects MLIRRegisterAllExtensions MLIRRegisterAllPasses)
  if(TARGET ${lib})
    list(APPEND NM_COMPILER_LIBS ${lib})
  endif()
endforeach()
add_llvm_executable(nm-opt nm-opt.cpp PrepareExecutors.cpp)
set_target_properties(nm-opt PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/compiler/bin")
add_dependencies(nm-opt nm_generated)
llvm_update_compile_flags(nm-opt)
target_link_libraries(nm-opt PRIVATE ${NM_COMPILER_LIBS})
mlir_check_all_link_libraries(nm-opt)
